class Monitor:
    def reset(self):
        self.trace = []

    def step(self, propositions):
        self.trace.append(set(propositions))

    def finish(self):
        n = len(self.trace)

        # Rule 1: After every visit to A, reach B 1-3 steps later
        for i in range(n):
            if 'A' not in self.trace[i]:
                continue
            window = range(i + 1, min(i + 4, n))
            if not any('B' in self.trace[j] for j in window):
                return False

        # Rule 2: After every visit to B, avoid X until C is reached
        for i in range(n):
            if 'B' not in self.trace[i]:
                continue
            ok = False
            for k in range(i, n):
                if 'C' not in self.trace[k]:
                    continue
                if all('X' not in self.trace[m] for m in range(i + 1, k)):
                    ok = True
                    break
            if not ok:
                return False

        return True
