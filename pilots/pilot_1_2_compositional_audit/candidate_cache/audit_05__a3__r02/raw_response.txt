class Monitor:
    def reset(self):
        self.trace = []

    def step(self, propositions):
        self.trace.append(set(propositions))

    def finish(self):
        n = len(self.trace)

        # A must be visited at least once
        has_a = any('A' in props for props in self.trace)
        if not has_a:
            return False

        # For every visit to A, check the requirement
        for i in range(n):
            if 'A' not in self.trace[i]:
                continue

            # Must reach B in the next 1-2 steps
            satisfied = False
            for j in range(i + 1, min(i + 3, n)):
                if 'B' not in self.trace[j]:
                    continue
                # After this B, must eventually reach C at or after B
                if any('C' in self.trace[k] for k in range(j, n)):
                    satisfied = True
                    break

            if not satisfied:
                return False

        return True
