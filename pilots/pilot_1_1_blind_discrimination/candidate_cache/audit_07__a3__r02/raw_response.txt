```python
class Monitor:
    def reset(self) -> None:
        self.a_visited = False
        self.b_visited = False
        self.c_visited = False
        self.x_visiting = False

    def step(self, propositions: set[str]) -> None:
        if 'A' in propositions and not self.a_visited:
            self.a_visited = True
            self.c_visited = False  # reset c_visiting
        elif 'B' in propositions and not self.b_visited:
            self.b_visited = True
            self.x_visiting = False  # reset x_visiting
        elif 'C' in propositions and not self.c_visited:
            self.c_visited = True
            if 'A' not in propositions or ('X' in propositions and self.x_visiting):
                return
            else:
                self.a_visited = False  # clear a_visited
                self.b_visited = False  # clear b_visited

    def finish(self) -> bool:
        return self.c_visited
```
