import random

class Monitor:
    def __init__(self):
        self.propositions = set()
        self.traces = []

    def reset(self) -> None:
        self.propositions = set()
        self.traces = []

    def step(self, propositions: set[str]) -> None:
        for proposition in propositions:
            if proposition == 'A':
                self.propositions.add('A')
                self.add_trace(0)
                self.add_trigger('D', 1, 3)
            elif proposition == 'B':
                self.propositions.add('B')
                self.add_trace(len(self.traces))
                self.add_trigger('C', len(self.traces), len(self.traces) + 2)
                if self.get_last_visited_prop() == 'A':
                    self.add_trigger('D', random.randint(1, 3), len(self.traces))
            elif proposition == 'X' and not self.get_last_visited_prop():
                self.propositions.remove('X')
                if self.get_last_visited_prop() == 'B':
                    self.add_trigger('C', len(self.traces), len(self.traces) + 2)
            else:
                continue

    def finish(self) -> bool:
        return len(self.traces) > 0
