from warehouse_dsl import parse_task, canonicalize, evaluate_task

def task():
    return "all(after_each(A, between(2, 4, visit(C))), after_each(B, between(1, 2, visit(D))), avoid(X))"

tree = parse_task(task())
text = canonicalize(task())
ok = evaluate_task(
    task(),
    (frozenset(), frozenset({"A", "B", "C", "D"})),
)

print("task =", tree)
print("canonicalized_task =", text)
print("result =", ok)
