from warehouse_dsl import parse_task, canonicalize, evaluate_task

def visit_a_before_b_and_c_and_d(task):
    return (
        all(
            after_each(A, between(1, 3, C)),
            after_each(B, between(2, 4, D))
        )
        for A in [step for step in task.steps if step.label == 'A']
        for B in [step for step in task.steps if step.label == 'B']
    )

task = {
    'steps': [
        {'label': 'A'},
        {'label': 'B'}
    ]
}

tree = parse_task("visit(A) and after_each(B, between(2, 4, D))")
text = canonicalize(" visit(A) and after_each(B,between(2,4,D) ) ")
ok = evaluate_task(
    "visit(A) and after_each(B, between(2, 4, D))",
    (frozenset([{'label': 'A'}]), frozenset([{'label': 'B'}, {'label': 'D'}]))
)

print(tree)
print(text)
print(ok)
