from warehouse_dsl import parse_task, canonicalize, evaluate_task

def task():
    return sequence(
        visit(A),
        between(1, 3, visit(D)),
        after_each(visit(B), avoid_until(X, D))
    )

task_ = parse_task(task())
canonicalized_task_ = canonicalize(task_)
ok = evaluate_task(canonicalized_task_, (frozenset(), frozenset({"A"}), frozenset({"D"}), frozenset({"B"}), frozenset({"X"}), frozenset({"C"})))
print(ok)
