from warehouse_dsl import parse_task, canonicalize, evaluate_task

def audit_task():
    tree = parse_task("after_each(A, avoid_until(X, C))")
    text = canonicalize(" after_each ( A,avoid_until(X,C) ) ")
    ok = evaluate_task(
        "after_each(A, avoid_until(X, C))",
        ((frozenset(), frozenset({"A", "X", "C"})),
         (frozenset({"A", "C"}), frozenset({"X"})))
    )
    return tree
