```python
class Monitor:
    def __init__(self):
        self.a_visited = False
        self.b_visited = False
        self.c_visited = False
        self.x_forbidden = True

    def reset(self) -> None:
        self.a_visited = False
        self.b_visited = False
        self.c_visited = False
        self.x_forbidden = True

    def step(self, propositions: set[str]) -> None:
        if 'A' in propositions:
            self.a_visited = True
        elif 'B' in propositions and self.a_visited:
            self.b_visited = True
            self.x_forbidden = False
        elif 'C' in propositions and self.a_visited:
            self.c_visited = True

    def finish(self) -> bool:
        return self.b_visited or self.c_visited
```
