class Monitor:
    def reset(self) -> None:
        self.a_visited = False
        self.b_visited = False
        self.c_visited = False
        self.x_avoided = True
        self.a_offset = 0
        self.b_offset = 0
        self.c_offset = 0

    def step(self, propositions: set[str]) -> None:
        if 'A' in propositions and not self.a_visited:
            self.a_visited = True
            self.a_offset += 1
            if self.a_offset < 2 or self.a_offset > 4:
                raise ValueError("Invalid A offset")
        elif 'B' in propositions and self.a_visited:
            self.b_visited = True
            self.b_offset += 1
            if self.b_offset < 1 or self.b_offset > 2:
                raise ValueError("Invalid B offset")
        elif 'C' in propositions and (self.a_visited or self.b_visited):
            self.c_visited = True
            self.c_offset += 1
            if not self.x_avoided:
                raise ValueError("X is present before C")
            if self.c_offset < 2 or self.c_offset > 4:
                raise ValueError("Invalid C offset")

    def finish(self) -> bool:
        return (self.a_visited and self.b_visited and self.c_visited) and all([
            (not self.x_avoided),
            (self.c_offset >= 1 and self.c_offset <= 2)
        ])
