class Monitor:
    def reset(self) -> None:
        self.a_visited = False
        self.b_visited = False
        self.c_visited = {}
        self.d_visited = {}

    def step(self, propositions: set[str]) -> None:
        if 'A' in propositions:
            self.a_visited = True
            self.c_visited[len(propositions) - 1] = True

        elif 'B' in propositions and not self.b_visited:
            self.b_visited = True
            self.d_visited[len(propositions) - 2] = True

    def finish(self) -> bool:
        return self.a_visited and self.b_visited and all(self.c_visited.values()) and all(self.d_visited.values())
