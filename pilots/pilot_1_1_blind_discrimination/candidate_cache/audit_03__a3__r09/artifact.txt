class Monitor:
    def reset(self) -> None:
        self.a_visited = False
        self.b_visited = False
        self.current_step = 0

    def step(self, propositions: set[str]) -> None:
        if 'A' in propositions and not self.a_visited:
            self.a_visited = True
            self.current_step += 1
        elif 'B' in propositions and not self.b_visited:
            self.b_visited = True
            if self.current_step > 1:
                self.current_step += 1

    def finish(self) -> bool:
        return (self.a_visited and self.b_visited and 
                self.current_step == len(set('A B'))) or 
               (not self.a_visited and not self.b_visited)
