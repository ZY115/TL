from warehouse_dsl import parse_task, canonicalize, evaluate_task
import re
from dataclasses import dataclass
from typing import Tuple

# Define a custom value class to represent the anchor step.
@dataclass(frozen=True)
class Anchor:
    k: int = 0

# Custom task parser for the current audit task. It's not included in the standard DSL grammar, so we need to define our own.
def parse_task_current_audit(source: str) -> Tuple[Anchor]:
    # Split source into individual parts (after each A).
    after_a_parts = re.split(r'\s*(A)\s*', source)

    anchor = Anchor(k=0)
    for i in range(1, len(after_a_parts), 2):
        # Assign an anchor to every part that contains 'A'.
        parts_with_a = [part for part in after_a_parts[i - 1:i + 1] if re.search(r'\w', part)]
        if not parts_with_a:
            continue
        anchor.k += 1

    return anchor, parse_task(source)

# Custom task evaluation function.
def evaluate_task_current_audit(task: str) -> bool:
    # Split source into individual parts (after each A).
    after_a_parts = re.split(r'\s*(A)\s*', task)
    
    for i in range(1, len(after_a_parts), 2):
        # Assign an anchor to every part that contains 'A'.
        parts_with_a = [part for part in after_a_parts[i - 1:i + 1] if re.search(r'\w', part)]
        
        # Evaluate the task at each anchor.
        result = True
        for part in parts_with_a:
            if not re.search(r'\bX\b', part):
                continue

            if i == len(after_a_parts) - 1 or not re.search(r'C$', after_a_parts[i + 1]):
                return False
        
        # Return the result.
        result = any(re.search(r'B$', part) for part in parts_with_a)
    
    return True
